# Example: Full Skinaiq scan import requests response = requests.post("http://localhost:5000/api/scan/full", files={ "image": open("skin_photo.jpg", "rb"), "audio": open("voice_sample.wav", "rb"), }, data={"biometrics": '{"hr":72,"hrv":62,"spo2":98}'}) report = response.json() print(f"Health Score: {report['summary']['health_score']}") print(f"Top Risk: {report['summary']['top_risk']['name']}") print(f"Confidence: {report['summary']['top_risk']['confidence']}%")
